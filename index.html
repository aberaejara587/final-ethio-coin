using System; using System.Globalization; using System.Windows.Forms; namespace AberaCalculator1 { public partial class Form1 : Form { private double _accumulator = 0.0; private string _pendingOperator = string.Empty; private bool _isNewEntry = true; public Form1() { InitializeComponent(); // Initialize display txtDisplay.Text = "0"; // Wire up numeric buttons btn0.Click += Digit_Click; btn1.Click += Digit_Click; btn2.Click += Digit_Click; btn3.Click += Digit_Click; btn4.Click += Digit_Click; btn5.Click += Digit_Click; btn6.Click += Digit_Click; btn7.Click += Digit_Click; btn8.Click += Digit_Click; btn9.Click += Digit_Click; // Other buttons btnDot.Click += BtnDot_Click; btnClear.Click += BtnClear_Click; btnBackspace.Click += BtnBackspace_Click; btnPercent.Click += BtnPercent_Click; // Operators btnPlus.Click += Operator_Click; btnMinus.Click += Operator_Click; btnMultiply.Click += Operator_Click; btnDivide.Click += Operator_Click; btnEquals.Click += BtnEquals_Click; } private void Digit_Click(object? sender, EventArgs e) { if (sender is not Button btn) return; if (_isNewEntry || txtDisplay.Text == "0") { txtDisplay.Text = btn.Text; _isNewEntry = false; } else { txtDisplay.Text += btn.Text; } } private void BtnDot_Click(object? sender, EventArgs e) { if (_isNewEntry) { txtDisplay.Text = "0."; _isNewEntry = false; return; } if (!txtDisplay.Text.Contains(".")) txtDisplay.Text += "."; } private void Operator_Click(object? sender, EventArgs e) { if (sender is not Button btn) return; try { double current = double.Parse(txtDisplay.Text, CultureInfo.InvariantCulture); if (!string.IsNullOrEmpty(_pendingOperator) && !_isNewEntry) { // perform previous operation _accumulator = Calculate(_accumulator, current, _pendingOperator); txtDisplay.Text = _accumulator.ToString(CultureInfo.InvariantCulture); } else { _accumulator = current; } _pendingOperator = btn.Text; _isNewEntry = true; } catch { txtDisplay.Text = "Error"; _isNewEntry = true; _pendingOperator = string.Empty; _accumulator = 0.0; } } private void BtnEquals_Click(object? sender, EventArgs e) { if (string.IsNullOrEmpty(_pendingOperator)) return; try { double second = double.Parse(txtDisplay.Text, CultureInfo.InvariantCulture); double result = Calculate(_accumulator, second, _pendingOperator); txtDisplay.Text = result.ToString(CultureInfo.InvariantCulture); // Reset state so user can continue from result _accumulator = result; _pendingOperator = string.Empty; _isNewEntry = true; } catch { txtDisplay.Text = "Error"; _isNewEntry = true; _pendingOperator = string.Empty; _accumulator = 0.0; } } private void BtnClear_Click(object? sender, EventArgs e) { txtDisplay.Text = "0"; _accumulator = 0.0; _pendingOperator = string.Empty; _isNewEntry = true; } private void BtnBackspace_Click(object? sender, EventArgs e) { if (_isNewEntry) { txtDisplay.Text = "0"; return; } if (txtDisplay.Text.Length <= 1) { txtDisplay.Text = "0"; _isNewEntry = true; } else { txtDisplay.Text = txtDisplay.Text[..^1]; } } private void BtnPercent_Click(object? sender, EventArgs e) { try { double value = double.Parse(txtDisplay.Text, CultureInfo.InvariantCulture); value = value / 100.0; txtDisplay.Text = value.ToString(CultureInfo.InvariantCulture); _isNewEntry = true; } catch { txtDisplay.Text = "Error"; _isNewEntry = true; } } private static double Calculate(double left, double right, string op) { return op switch { "+" => left + right, "-" => left - right, "*" => left * right, "/" => right == 0 ? throw new DivideByZeroException() : left / right, _ => right }; } private void txtDisplay_TextChanged(object sender, EventArgs e) { // kept intentionally empty - UI update only } } }
